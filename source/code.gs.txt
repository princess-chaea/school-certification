// --- doGet 함수: 웹 앱의 초기 로드 및 GET 요청 처리 (수정됨) ---
function doGet(e) {
  const params = e.parameter;

  // API 요청 처리 (GET 요청으로 데이터를 반환)
  if (params.getAllStudentStats !== undefined) return getAllStudentStats();
  if (params.getStudentStats !== undefined) return getStudentStats(params.name);
  if (params.getMissions !== undefined) {
    const studentName = params.name;
    return getMissions(studentName); 
  }
  if (params.getStudentSubmissions !== undefined) return getStudentSubmissions(params.name);
  if (params.getStudentPointRequests !== undefined) return getStudentPointRequests(params.name); 
  if (params.getPendingRequests !== undefined) return getPendingRequests(params.teacher); // 교사 페이지 대기 요청
  if (params.getNotices !== undefined) return getNotices(); 
  if (params.getStudentImageMap !== undefined) return getStudentImageMap(); 
  if (params.getPointUsageItems !== undefined) return getPointUsageItems();
  if (params.getSchoolLogoUrl !== undefined) return getSchoolLogoUrl(); 
  if (params.getTeacherStudentsStats !== undefined) return getTeacherStudentsStats(params.teacher); // 교사 대시보드용 학생 통계
  if (params.getTeacherCategoryCounts !== undefined) return getTeacherCategoryCounts(params.teacher);
  if (params.getTeacherMonthlyPoints !== undefined) return getTeacherMonthlyPoints(params.teacher);
  if (params.getLatestFeedbackForStudent !== undefined) {
    const name = params.studentName || params.name;
    const grade = params.studentGrade || params.grade;
    return getLatestFeedbackForStudent(name, grade);
  }
  if (params.getCertificationExplanations !== undefined) return getCertificationExplanations(); 
  
  // ⭐⭐ 이모지 목록 요청 처리 (추가) ⭐⭐
  if (params.getEmojis !== undefined) return getEmojis();

  // HTML 페이지 요청 처리
  if (params.page === 'certification_explanation') { 
      const template = HtmlService.createTemplateFromFile("certification_explanation");
      template.WEB_APP_URL = ScriptApp.getService().getUrl(); 
      return template.evaluate().setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  if (params.page === 'teacher' && params.teacher) { 
    const template = HtmlService.createTemplateFromFile("teacher");
    template.teacher = params.teacher;
    return template.evaluate().setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  if (params.name && params.grade) {
    logStudentLogin(params.name);
    const template = HtmlService.createTemplateFromFile("index");
    template.name = params.name;
    template.grade = params.grade;
    template.WEB_APP_URL = ScriptApp.getService().getUrl(); 
    return template.evaluate().setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  return HtmlService.createHtmlOutputFromFile("main").setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// --- doPost 함수: 웹 앱의 POST 요청 처리 (통합 버전) ---
function doPost(e) {
  try {
    const params = JSON.parse(e.postData.contents);

    if (typeof params !== 'object' || params === null) {
      return jsonResponse({ status: "error", message: "요청 데이터 파싱 오류" });
    }

    switch (params.action) {
      case 'submitMission':
        return submitMission({
          name: params.name,
          grade: params.grade,
          category: params.category,
          mission: params.mission,
          imageUrl: params.imageUrl
        });
      case 'requestPoint':
        return requestPoint(params);

      // ⭐⭐ 이모지 구매 처리 (추가) ⭐⭐
      case 'addEmoji':
        return addEmoji(params); // 이전에 추가한 addEmoji 함수를 호출합니다.

      case 'approvePointRequest':
        return approvePointRequest(params);
      case 'uploadImage':
        if (!params.name || !params.base64) {
          return jsonResponse({ status: "error", message: "업로드 필수 데이터 누락" });
        }
        return uploadImage(params);
      case 'saveStudentFeedback':
        return saveStudentFeedback({
          studentName: params.studentName,
          studentGrade: params.studentGrade,
          feedbackText: params.feedbackText,
          teacherName: params.teacherName
        });
      default:
        return jsonResponse({ status: "error", message: `알 수 없는 요청 action: ${params.action}` });
    }
  } catch (error) {
    Logger.log("BACKEND ERROR (doPost - catch): " + error.stack);
    return jsonResponse({ status: "error", message: "서버 내부 오류: " + error.message });
  }
}

// --- doOptions 함수: CORS Preflight 요청 처리 (수정된 버전) ---
function doOptions(e) {
  return ContentService.createTextOutput()
    .setStatusCode(204) // ⭐ Preflight 요청 성공을 의미하는 '204 No Content' 상태 코드 설정
    .withHeaders({ // 헤더를 한 번에 설정하는 것이 더 깔끔합니다.
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type'
    });
}

function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- formatKoreanDateTime 함수: 날짜/시간 형식 변환 유틸리티 ---
function formatKoreanDateTime(date) {
  if (!date || isNaN(new Date(date))) {
    return "-";
  }
  const d = new Date(date);
  return Utilities.formatDate(d, "Asia/Seoul", "yyyy. M. d. HH:mm");
}

// --- removeStudentPrefix 함수: '나만의' 미션 접두사 제거 유틸리티 ---
function removeStudentPrefix(rawContent, studentName) {
  if (rawContent && studentName && String(rawContent).startsWith(String(studentName) + "_")) {
    return String(rawContent).substring(String(studentName).length + 1);
  }
  return rawContent;
}

    if (!studentSheet || !submissionSheet || !loginSheet || !pointRequestSheet) {
        return jsonResponse({ error: "필수 시트를 찾을 수 없습니다." });
    }

    const allStudentsData = studentSheet.getDataRange().getValues().slice(1);
    const studentInfo = allStudentsData.find(row => String(row[2]).trim() === String(studentName).trim());

    if (!studentInfo) {
        return jsonResponse({ error: "학생을 찾을 수 없습니다." });
    }

    const gradeStudentList = String(studentInfo[0]).trim();

    const pointRequestData = pointRequestSheet.getDataRange().getValues().slice(1);

    const availablePoints = totalEarnedPoints - totalUsedPoints;

    const logins = loginSheet.getDataRange().getValues().slice(1);

    const gradedSubmissions = allStudentSubmissions.filter(r => ["금", "은", "동"].includes(String(r[6]).trim()));
    const goldCount = gradedSubmissions.filter(r => String(r[6]).trim() === '금').length;
    const silverCount = gradedSubmissions.filter(r => String(r[6]).trim() === '은').length;
    const bronzeCount = gradedSubmissions.filter(r => String(r[6]).trim() === '동').length;
    const myLogins = logins.filter(r => String(r[1]).trim() === nameStudentList);

    const studentStats = {
        name: nameStudentList,
        grade: gradeStudentList,
        gold: goldCount,
        silver: silverCount,
        bronze: bronzeCount,
        submissionCount: allStudentSubmissions.length,
        loginCount: myLogins.length,
        totalPoint: totalEarnedPoints,
        usedPoints: totalUsedPoints,
        availablePoints: availablePoints,
        // ⭐⭐ 학생 목록 시트의 6번째 열(F열)에서 이모지 정보를 가져옵니다. ⭐⭐
        emoji: studentInfo[5] || "" 
    };

    return jsonResponse(studentStats);
}

// --- getStudentTotalPoints 함수: 특정 학생의 총 획득 포인트 계산 (제출/로그인 기록 기반) ---
// 목적: '제출 내역'과 '로그인기록' 시트를 기반으로 학생의 총 획득 포인트를 계산
function getStudentTotalPoints(studentName) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet();
  const submissionSheet = sheet.getSheetByName("제출 내역");
  const loginSheet = sheet.getSheetByName("로그인기록");

  if (!submissionSheet || !loginSheet) {
    Logger.log("오류: '제출 내역' 또는 '로그인기록' 시트를 찾을 수 없어 포인트 계산이 불가합니다.");
    return 0;
  }

  const pointSettings = getPointSettings(); // getPointSettings 함수는 존재해야 함
  const POINT_GOLD = pointSettings.금 || 5;
  const POINT_SILVER = pointSettings.은 || 3;
  const POINT_BRONZE = pointSettings.동 || 1;
  const POINT_LOGIN = pointSettings.로그인 || 1;
  const POINT_SUBMISSION_PER_COUNT = pointSettings.제출 || 1;

  const submissions = submissionSheet.getDataRange().getValues().slice(1);
  const allStudentSubmissions = submissions.filter(r => String(r[1]).trim() === String(studentName).trim());

  const gradedSubmissions = allStudentSubmissions.filter(r =>
      ["금", "은", "동"].includes(String(r[6]).trim())
  );

  const goldCount = gradedSubmissions.filter(r => String(r[6]).trim() === '금').length;
  const silverCount = gradedSubmissions.filter(r => String(r[6]).trim() === '은').length;
  const bronzeCount = gradedSubmissions.filter(r => String(r[6]).trim() === '동').length;

  const goldPoints = goldCount * POINT_GOLD;
  const silverPoints = silverCount * POINT_SILVER;
  const bronzePoints = bronzeCount * POINT_BRONZE;

  const submissionCountPoints = allStudentSubmissions.length * POINT_SUBMISSION_PER_COUNT;

  const logins = loginSheet.getDataRange().getValues().slice(1);
  const myLogins = logins.filter(r => String(r[1]).trim() === String(studentName).trim());
  const loginCountPoints = myLogins.length * POINT_LOGIN;

  const totalEarnedPoints = goldPoints + silverPoints + bronzePoints + submissionCountPoints + loginCountPoints;

  Logger.log(`학생 ${studentName}의 총 획득 포인트 (계산): ${totalEarnedPoints}`);
  return totalEarnedPoints;
}

/**
 * =================================================================================
 * 🚀 성능 개선 버전: 모든 학생의 통계를 효율적으로 계산합니다.
 * - 각 데이터 시트를 단 한 번만 읽고, 메모리에서 데이터를 가공하여 처리 속도를 향상시킵니다.
 * =================================================================================
 */
function getAllStudentStats() {
    const sheet = SpreadsheetApp.getActiveSpreadsheet();
    const studentSheet = sheet.getSheetByName("학생 목록");
    const submissionSheet = sheet.getSheetByName("제출 내역");
    const loginSheet = sheet.getSheetByName("로그인기록");
    const pointRequestSheet = sheet.getSheetByName("포인트사용요청");

    // 1. 필수 시트 확인
    if (!studentSheet || !submissionSheet || !loginSheet || !pointRequestSheet) {
        return ContentService.createTextOutput(JSON.stringify({ error: "필수 시트를 찾을 수 없습니다." }))
            .setMimeType(ContentService.MimeType.JSON);
    }

    // 2. 각 시트의 전체 데이터를 메모리로 한 번에 로드합니다.
    const allStudentsData = studentSheet.getDataRange().getValues().slice(1);
    const allSubmissionsData = submissionSheet.getLastRow() > 1 ? submissionSheet.getDataRange().getValues().slice(1) : [];
    const allLoginsData = loginSheet.getLastRow() > 1 ? loginSheet.getDataRange().getValues().slice(1) : [];
    const allPointRequestData = pointRequestSheet.getDataRange().getValues().slice(1);

    // 3. 학생 이름(key)을 기준으로 데이터를 미리 가공하여 객체(map)에 저장합니다.
    // 이렇게 하면 학생을 순회할 때마다 시트 전체를 다시 검색할 필요가 없어집니다.
    const submissionsByStudent = {};
    const loginsByStudent = {};
    const approvedUsageByStudent = {};
    const loggedInTodayStudentNames = new Set();
    const todayString = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd");

    allSubmissionsData.forEach(row => {
        const name = String(row[1]).trim(); // 제출 내역 시트의 '이름' 열
        if (!submissionsByStudent[name]) submissionsByStudent[name] = [];
        submissionsByStudent[name].push(row);
    });

    allLoginsData.forEach(row => {
        const name = String(row[1]).trim(); // 로그인기록 시트의 '이름' 열
        if (!loginsByStudent[name]) loginsByStudent[name] = [];
        loginsByStudent[name].push(row);

        // 오늘 로그인 여부도 여기서 함께 확인합니다.
        const loginDate = new Date(row[2]);
        if (Utilities.formatDate(loginDate, "Asia/Seoul", "yyyy-MM-dd") === todayString) {
            loggedInTodayStudentNames.add(name);
        }
    });

    allPointRequestData.forEach(row => {
        if (String(row[4]).trim() === "승인") { // '상태' 열이 '승인'인 경우
            const name = String(row[0]).trim(); // 포인트사용요청 시트의 '이름' 열
            const points = Number(row[2]) || 0;
            approvedUsageByStudent[name] = (approvedUsageByStudent[name] || 0) + points;
        }
    });

    // 포인트 설정은 한 번만 가져옵니다.
    const pointSettings = getPointSettings();

    // 4. 학생 목록을 순회하며 미리 가공된 데이터를 사용하여 최종 통계를 조합합니다.
    const result = allStudentsData.map(studentInfo => {
        const nameStudentList = String(studentInfo[2]).trim();
        const gradeStudentList = String(studentInfo[0]).trim();

        // 미리 가공된 학생별 데이터를 가져옵니다. 데이터가 없으면 빈 배열을 사용합니다.
        const studentSubmissions = submissionsByStudent[nameStudentList] || [];
        const studentLogins = loginsByStudent[nameStudentList] || [];
        
        // ❗️ 시트를 다시 읽는 대신, 메모리의 데이터로 포인트를 계산하는 도우미 함수를 호출합니다.
        const { totalEarnedPoints, goldCount, silverCount, bronzeCount } = calculatePointsFromAggregatedData(studentSubmissions, studentLogins, pointSettings);
        
        const studentApprovedUsedPoints = approvedUsageByStudent[nameStudentList] || 0;
        const availablePoints = totalEarnedPoints - studentApprovedUsedPoints;
        const hasLoggedIn = loggedInTodayStudentNames.has(nameStudentList);

        return {
            name: nameStudentList,
            grade: gradeStudentList,
            gold: goldCount,
            silver: silverCount,
            bronze: bronzeCount,
            submissionCount: studentSubmissions.length,
            loginCount: studentLogins.length,
            totalPoint: totalEarnedPoints,
            usedPoints: studentApprovedUsedPoints,
            availablePoints: availablePoints,
            hasLoggedIn: hasLoggedIn,
            emoji: studentInfo[5] || ""
        };
    });

    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}


/**
 * =================================================================================
 * 🛠️ 도우미 함수: 미리 집계된 데이터로 학생의 총 획득 포인트와 메달 수를 계산합니다.
 * - 이 함수는 스프레드시트를 직접 읽지 않아 매우 빠릅니다.
 * =================================================================================
 */
function calculatePointsFromAggregatedData(submissions, logins, pointSettings) {
    const POINT_GOLD = pointSettings.금 || 5;
    const POINT_SILVER = pointSettings.은 || 3;
    const POINT_BRONZE = pointSettings.동 || 1;
    const POINT_LOGIN = pointSettings.로그인 || 1;
    const POINT_SUBMISSION_PER_COUNT = pointSettings.제출 || 1;

    const gradedSubmissions = submissions.filter(r => ["금", "은", "동"].includes(String(r[6]).trim()));
    const goldCount = gradedSubmissions.filter(r => String(r[6]).trim() === '금').length;
    const silverCount = gradedSubmissions.filter(r => String(r[6]).trim() === '은').length;
    const bronzeCount = gradedSubmissions.filter(r => String(r[6]).trim() === '동').length;

    const goldPoints = goldCount * POINT_GOLD;
    const silverPoints = silverCount * POINT_SILVER;
    const bronzePoints = bronzeCount * POINT_BRONZE;
    const submissionCountPoints = submissions.length * POINT_SUBMISSION_PER_COUNT;
    const loginCountPoints = logins.length * POINT_LOGIN;

    const totalEarnedPoints = goldPoints + silverPoints + bronzePoints + submissionCountPoints + loginCountPoints;

    return { totalEarnedPoints, goldCount, silverCount, bronzeCount };
}

// --- getTeacherStudentsStats 함수: 특정 교사의 담당 학생들 전체 통계 반환 ---
function getTeacherStudentsStats(teacherName) {
    if (!teacherName) {
        return ContentService.createTextOutput(JSON.stringify({ error: "교사 이름이 필요합니다." })).setMimeType(ContentService.MimeType.JSON);
    }

    const sheet = SpreadsheetApp.getActiveSpreadsheet();
    const studentSheet = sheet.getSheetByName("학생 목록"); // 학년(0), 반(1), 이름(2), 파일ID(3), 담임(4)
    const submissionSheet = sheet.getSheetByName("제출 내역");
    const loginSheet = sheet.getSheetByName("로그인기록");
    const pointRequestSheet = sheet.getSheetByName("포인트사용요청");

    if (!studentSheet || !submissionSheet || !loginSheet || !pointRequestSheet) {
        return ContentService.createTextOutput(JSON.stringify({ error: "필수 시트를 찾을 수 없습니다." }))
                             .setMimeType(ContentService.MimeType.JSON);
    }

    const allStudentsData = studentSheet.getDataRange().getValues().slice(1);
    const allSubmissions = submissionSheet.getDataRange().getValues().slice(1);
    const allLogins = loginSheet.getDataRange().getValues().slice(1);
    const allPointRequestData = pointRequestSheet.getDataRange().getValues().slice(1);

    // 🔑 수정: 담임 필터링 인덱스를 studentRow[4] (E열)로 변경
    const studentsOfThisTeacher = allStudentsData.filter(studentRow =>
        String(studentRow[4]).trim() === String(teacherName).trim() // 담임 필터링
    );

    const resultStats = studentsOfThisTeacher.map(studentInfo => {
        // 🔑 수정: nameStudentList는 studentInfo[2] (이름), gradeStudentList는 studentInfo[0] (학년)
        const nameStudentList = String(studentInfo[2]).trim();
        const gradeStudentList = String(studentInfo[0]).trim();

        const totalEarnedPoints = getStudentTotalPoints(nameStudentList);

        const studentApprovedUsedPoints = allPointRequestData.filter(row =>
            String(row[0]).trim() === String(nameStudentList).trim() && String(row[4]).trim() === "승인"
        ).reduce((acc, cur) => acc + (Number(cur[2]) || 0), 0);

        const availablePoints = totalEarnedPoints - studentApprovedUsedPoints;

        const studentSubmissions = allSubmissions.filter(r => String(r[1]).trim() === String(nameStudentList).trim());
        const studentLogins = allLogins.filter(r => String(r[1]).trim() === String(nameStudentList).trim());

        const gradedSubmissions = studentSubmissions.filter(r =>
            ["금", "은", "동"].includes(String(r[6]).trim())
        );

        const goldCount = gradedSubmissions.filter(r => String(r[6]).trim() === '금').length;
        const silverCount = gradedSubmissions.filter(r => String(r[6]).trim() === '은').length;
        const bronzeCount = gradedSubmissions.filter(r => String(r[6]).trim() === '동').length;

        return {
            name: nameStudentList,
            grade: gradeStudentList,
            gold: goldCount,
            silver: silverCount,
            bronze: bronzeCount,
            totalSubmissionCount: studentSubmissions.length,
            loginCount: studentLogins.length,
            totalPoint: totalEarnedPoints,
            usedPoints: studentApprovedUsedPoints,
            availablePoints: availablePoints,
        };
    });

    resultStats.sort((a, b) => b.totalPoint - a.totalPoint);

    return ContentService.createTextOutput(JSON.stringify(resultStats)).setMimeType(ContentService.MimeType.JSON);
}

// --- getTeacherCategoryCounts 함수: 특정 교사 담당 학생들의 카테고리별 인증 횟수 반환 ---

    // 🔑 수정: 학생 이름은 studentRow[2] (C열)에서 가져옴
    const studentsOfThisTeacherNames = allStudentsData.filter(studentRow =>
        String(studentRow[4]).trim() === String(teacherName).trim()
    ).map(studentRow => String(studentRow[2]).trim());

    Logger.log("Students of this teacher (for category counts): " + JSON.stringify(studentsOfThisTeacherNames));

    const categoryCounts = {};

    allSubmissions.forEach(subRow => {
        const studentNameInSubmission = String(subRow[1]).trim(); // 제출 내역 시트의 학생 이름 (B열)

        if (studentsOfThisTeacherNames.includes(studentNameInSubmission)) {
            const category = String(subRow[2] || "").trim(); // 사업명 (인덱스 2)
            if (category) {
                categoryCounts[category] = (categoryCounts[category] || 0) + 1;
            }
        }
    });

    Logger.log("Final categoryCounts: " + JSON.stringify(categoryCounts));
    return ContentService.createTextOutput(JSON.stringify(categoryCounts)).setMimeType(ContentService.MimeType.JSON);
}

// --- getTeacherMonthlyPoints 함수: 특정 교사 담당 학생들의 월별 총 획득 포인트 반환 ---
function getTeacherMonthlyPoints(teacherName) {
    if (!teacherName) {
        return ContentService.createTextOutput(JSON.stringify({ error: "교사 이름이 필요합니다." })).setMimeType(ContentService.MimeType.JSON);
    }

    const sheet = SpreadsheetApp.getActiveSpreadsheet();
    const studentSheet = sheet.getSheetByName("학생 목록"); // 학년(0), 반(1), 이름(2), 파일ID(3), 담임(4)
    const submissionSheet = sheet.getSheetByName("제출 내역");
    const loginSheet = sheet.getSheetByName("로그인기록");

    if (!studentSheet || !submissionSheet || !loginSheet) {
        return ContentService.createTextOutput(JSON.stringify({ error: "필수 시트를 찾을 수 없습니다." }))
                             .setMimeType(ContentService.MimeType.JSON);
    }

    const allStudentsData = studentSheet.getDataRange().getValues().slice(1);
    const allSubmissions = submissionSheet.getDataRange().getValues().slice(1);
    const allLogins = loginSheet.getDataRange().getValues().slice(1);

    const pointSettings = getPointSettings();
    const POINT_GOLD = pointSettings.금 || 5;
    const POINT_SILVER = pointSettings.은 || 3;
    const POINT_BRONZE = pointSettings.동 || 1;
    const POINT_LOGIN = pointSettings.로그인 || 1;
    const POINT_SUBMISSION_PER_COUNT = pointSettings.제출 || 1;

    // 🔑 수정: 담임 필터링 인덱스를 studentRow[4] (E열)로 변경
    // 🔑 수정: 학생 이름은 studentRow[2] (C열)에서 가져옴
    const studentsOfThisTeacherNames = allStudentsData.filter(studentRow =>
        String(studentRow[4]).trim() === String(teacherName).trim()
    ).map(studentRow => String(studentRow[2]).trim());

    const monthlyPoints = {};

    studentsOfThisTeacherNames.forEach(studentName => {
        const studentSubmissions = allSubmissions.filter(r => String(r[1]).trim() === studentName);
        const studentLogins = allLogins.filter(r => String(r[1]).trim() === studentName);

        studentSubmissions.forEach(sub => {
            const submittedDate = new Date(sub[5]);
            if (isNaN(submittedDate)) {
                Logger.log(`경고: ${studentName}의 제출 날짜 파싱 실패: ${sub[5]}`);
                return;
            }
            const monthKey = Utilities.formatDate(submittedDate, "Asia/Seoul", "yyyy-MM");
            const level = String(sub[6] || "").trim();
            let earned = 0;
            if (level === '금') earned += POINT_GOLD;
            else if (level === '은') earned += POINT_SILVER;
            else if (level === '동') earned += POINT_BRONZE;
            earned += POINT_SUBMISSION_PER_COUNT;

            monthlyPoints[monthKey] = (monthlyPoints[monthKey] || 0) + earned;
        });

        studentLogins.forEach(login => {
            const loginDate = new Date(login[2]);
            if (isNaN(loginDate)) {
                Logger.log(`경고: ${studentName}의 로그인 날짜 파싱 실패: ${login[2]}`);
                return;
            }
            const monthKey = Utilities.formatDate(loginDate, "Asia/Seoul", "yyyy-MM");
            monthlyPoints[monthKey] = (monthlyPoints[monthKey] || 0) + POINT_LOGIN;
        });
    });

    const finalMonthlyPointsArray = [];
    const today = new Date();
    for (let i = 11; i >= 0; i--) {
        const d = new Date(today.getFullYear(), today.getMonth() - i, 1);
        const monthKey = Utilities.formatDate(d, "Asia/Seoul", "yyyy-MM");
        finalMonthlyPointsArray.push({
            month: monthKey,
            points: monthlyPoints[monthKey] || 0
        });
    }

    return ContentService.createTextOutput(JSON.stringify(finalMonthlyPointsArray)).setMimeType(ContentService.MimeType.JSON);
}

// --- getMissions 함수: 모든 미션 목록을 '학교장 인증 항목' 시트에서 가져옴 ---
// 목적: 학생에게 보여줄 전체 미션 목록 (공통 + '나만의' 포함, '나만의'는 해당 학생의 것만 표시)
function getMissions(studentName) {
    const cache = CacheService.getScriptCache();
    const cacheKey = "missions_" + studentName;

    const cached = cache.get(cacheKey);
    if (cached) {
        Logger.log("✅ 캐시 사용중 for " + studentName);
        return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON);
    }

    Logger.log("🔄 시트 새로 읽기 for " + studentName);

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const missionSheet = ss.getSheetByName('학교장 인증 항목');
    const submissionSheet = ss.getSheetByName('제출 내역');

    let missionsData = [];
    let studentSubmissions = [];

    if (submissionSheet && submissionSheet.getLastRow() > 1) {
        const allSubmissions = submissionSheet.getDataRange().getValues().slice(1);
        studentSubmissions = allSubmissions.filter(row => String(row[1]).trim() === String(studentName).trim());
    }

    const missionStats = {};
    const levelPriority = { '금': 3, '은': 2, '동': 1, '': 0 };

    studentSubmissions.forEach(sub => {
        const missionContent = String(sub[3]).trim(); // '제출 내역' 시트의 '인증항목'
        // ⭐⭐ 미션 내용을 정규화하여 카운팅 키로 사용합니다. ⭐⭐
        const normalizedMissionContent = missionContent.replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
        Logger.log(`DEBUG (getMissions): 제출 내역에서 정규화된 미션: '${normalizedMissionContent}' (원본: '${missionContent}')`);
        
        const level = String(sub[6] || '').trim();

        if (!missionStats[normalizedMissionContent]) { // 정규화된 이름으로 키 사용
            missionStats[normalizedMissionContent] = { count: 0, highestLevel: '' };
        }
        missionStats[normalizedMissionContent].count = (missionStats[normalizedMissionContent].count || 0) + 1;

        if (levelPriority[level] > levelPriority[missionStats[normalizedMissionContent].highestLevel]) {
            missionStats[normalizedMissionContent].highestLevel = level;
        }
    });

    if (missionSheet && missionSheet.getLastRow() > 1) {
        const rawData = missionSheet.getRange(2, 1, missionSheet.getLastRow() - 1, 6).getValues();

        missionsData = rawData.map(row => {
            let content = String(row[2]).trim(); // '학교장 인증 항목' 시트의 '인증 항목 내용'
            const category = String(row[1]).trim();
            const rawMissionContent = String(row[2]).trim(); // 원본 미션 내용을 저장

            // ⭐⭐ '학교장 인증 항목' 시트의 미션 내용도 정규화하여 통계 조회 키로 사용합니다. ⭐⭐
            const normalizedMissionKey = rawMissionContent.replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
            Logger.log(`DEBUG (getMissions): 인증 항목에서 정규화된 미션 키: '${normalizedMissionKey}' (원본: '${rawMissionContent}')`);

            let currentStats = missionStats[normalizedMissionKey] || { count: 0, highestLevel: '' }; // 정규화된 이름으로 통계 조회

            if (category === '나만의') {
                if (rawMissionContent.startsWith(studentName + "_")) {
                    content = removeStudentPrefix(rawMissionContent, studentName); // 디스플레이용은 원본 유지
                    return {
                        category: category,
                        content: content, // 디스플레이용은 정규화하지 않음 (줄바꿈 반영은 프론트엔드에서)
                        gold: row[3], silver: row[4], bronze: row[5],
                        currentCount: currentStats.count,
                        highestLevel: currentStats.highestLevel
                    };
                } else {
                    return null;
                }
            } else {
                return {
                    category: category,
                    content: content, // 디스플레이용은 정규화하지 않음 (줄바꿈 반영은 프론트엔드에서)
                    gold: row[3], silver: row[4], bronze: row[5],
                    currentCount: currentStats.count,
                    highestLevel: currentStats.highestLevel
                };
            }
        }).filter(item => item !== null);
    }

    const json = JSON.stringify(missionsData);
    cache.put(cacheKey, json, 60 * 10);
    return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

// --- getStudentSubmissions 함수: 특정 학생의 제출 내역 반환 ---
function getStudentSubmissions(name) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("제출 내역");
  if (!sheet || sheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }

  const data = sheet.getDataRange().getValues().slice(1);
  // ✅ 여기서 각 열의 데이터를 올바른 인덱스에 매핑하고 있는지 확인
  // '제출 내역' 시트: 학년(0), 이름(1), 사업명(2), 인증항목(3), 사진URL(4), 제출시각(5), 등급(6)
  const filtered = data.filter(row => String(row[1]).trim() === String(name).trim()); // 이름은 B열 (인덱스 1)

  const result = filtered.map(row => {
    return {
      // ✅ '사업명'이 C열 (인덱스 2)에 있는지 확인
      category: String(row[2]).trim(), 
      // ✅ '인증항목'이 D열 (인덱스 3)에 있는지 확인
      mission: String(row[3]).trim(), 
      // ✅ '사진URL'이 E열 (인덱스 4)에 있는지 확인
      imageUrl: String(row[4]).trim(),
      // ✅ '제출시각'이 F열 (인덱스 5)에 있는지 확인
      submittedAt: Utilities.formatDate(new Date(row[5]), "Asia/Seoul", "yyyy. M. d. HH:mm"),
      // ✅ '등급'이 G열 (인덱스 6)에 있는지 확인
      level: String(row[6] || "").trim() 
    };
  });

  // 정렬 로직을 '사업명' > '인증항목' 오름차순으로 변경합니다.
  result.sort((a, b) => {
    const categoryCompare = a.category.localeCompare(b.category);
    if (categoryCompare !== 0) {
      return categoryCompare;
    }
    return a.mission.localeCompare(b.mission);
  });

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- submitMission 함수: 미션 제출 내역 기록 (수정됨) ---
function submitMission({ name, grade, category, mission, imageUrl }) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("제출 내역");
  if (!sheet) {
    Logger.log("오류: '제출 내역' 시트를 찾을 수 없습니다.");
    return jsonResponse({ status: "error", message: "'제출 내역' 시트를 찾을 수 없습니다. 관리자에게 문의하세요." });
  }

  const currentTime = new Date();
  const currentTimeString = Utilities.formatDate(currentTime, "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");
  const todayDateString = Utilities.formatDate(currentTime, "Asia/Seoul", "yyyy-MM-dd");

  // '나만의' 미션일 경우, 미션명에 학생 이름 접두사 추가를 먼저 수행합니다.
  if (category === "나만의" && !mission.startsWith(name + "_")) {
    mission = `${name}_${mission}`;
  }

  // 현재 제출하려는 미션 이름을 정규화합니다.
  const normalizedCurrentMission = String(mission).replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
  Logger.log(`DEBUG (submitMission): 현재 미션 정규화됨: '${normalizedCurrentMission}' (원본: '${mission}')`);


  // 2. 오늘 이미 제출된 미션인지 확인하는 로직
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const dataRange = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn());
    const values = dataRange.getValues();

    for (let i = 0; i < values.length; i++) {
      const row = values[i];
      const submittedName = String(row[1]).trim();
      const submittedCategory = String(row[2]).trim();
      const submittedMission = String(row[3]).trim();

      const normalizedSubmittedMission = submittedMission.replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
      Logger.log(`DEBUG (submitMission): 기존 제출 미션 정규화됨: '${normalizedSubmittedMission}' (원본: '${submittedMission}')`);

      const submittedTimeString = row[5];
      const submittedDateString = Utilities.formatDate(new Date(submittedTimeString), "Asia/Seoul", "yyyy-MM-dd");

      if (submittedName === name && submittedCategory === category && normalizedSubmittedMission === normalizedCurrentMission && submittedDateString === todayDateString) {
        Logger.log(`INFO: ${name} 학생이 오늘 '${mission}' 미션을 이미 제출했습니다. (카테고리: ${category})`);
        return jsonResponse({ status: "error", message: "이 미션은 오늘 이미 제출하셨습니다. 내일 다시 도전해주세요!" });
      }
    }
  }

  // ⭐⭐⭐ 핵심 수정: '나만의' 미션 개수 제한 확인 로직을 여기에서 제거합니다. ⭐⭐⭐
  // 이 로직은 addCustomMissionIfNeeded 함수로 완전히 이동했습니다.
  // 즉, 이 위치에서는 '나만의' 카테고리의 미션이더라도, 이미 존재하는 미션 카드라면 제출을 허용합니다.
  
  // '학교장 인증 항목' 시트에 '나만의' 미션이 없으면 추가
  // addCustomMissionIfNeeded 함수에서 3개 제한을 확인하고, 초과 시 오류를 throw 합니다.
  addCustomMissionIfNeeded(name, category, mission); 

  sheet.appendRow([grade, name, category, mission, imageUrl || "", currentTimeString, ""]);

  Logger.log(`'${name}'의 '${mission}' 제출 내역이 기록되었습니다.`);
  return jsonResponse({ status: "success", message: "미션이 성공적으로 제출되었습니다!" });
}


// --- addCustomMissionIfNeeded 함수: '나만의' 미션 자동 추가 (수정됨) ---
function addCustomMissionIfNeeded(name, category, mission) {
  if (category !== "나만의") return;

  const missionSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학교장 인증 항목");
  if (!missionSheet) {
    Logger.log("오류: '학교장 인증 항목' 시트를 찾을 수 없습니다. 미션 추가 불가.");
    // Browser.msgBox는 서버 사이드에서 작동하지 않습니다. 대신 오류를 던집니다.
    throw new Error("'학교장 인증 항목' 시트를 찾을 수 없습니다.");
  }

  const data = missionSheet.getDataRange().getValues();
  const existingNormalizedMissions = new Set(data.map(r => String(r[2]).trim().replace(/\n/g, ' ').replace(/\s+/g, ' ').trim()));
  Logger.log(`DEBUG (addCustomMissionIfNeeded): 기존 정규화된 미션 목록: ${Array.from(existingNormalizedMissions).join(', ')}`);

  const normalizedMissionToAdd = String(mission).trim().replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
  Logger.log(`DEBUG (addCustomMissionIfNeeded): 추가하려는 미션 정규화됨: '${normalizedMissionToAdd}' (원본: '${mission}')`);

  // ⭐⭐ '학교장 인증 항목'에 새로운 '나만의' 미션을 추가하려는 경우에만 제한 확인 ⭐⭐
  if (!existingNormalizedMissions.has(normalizedMissionToAdd)) { // 이 미션이 학교장 인증 항목에 아직 없는 경우 (새로운 나만의 미션)
    const currentCustomCount = getCurrentCustomMissionsCount(name); // 현재 학생의 '나만의' 미션 개수 확인
    if (currentCustomCount >= 3) {
      Logger.log(`INFO: '${name}' 학생의 '나만의' 미션 개수 초과로 '${mission}' 추가가 거부되었습니다. (현재 ${currentCustomCount}개)`);
      // ⭐ 개수 초과 시 오류를 던져서 submitMission에서 처리하게 함 ⭐
      throw new Error(`더 이상 '${name}' 학생의 '나만의' 미션을 추가할 수 없습니다. (최대 3개)`);
    }

    missionSheet.appendRow([
      missionSheet.getLastRow(), // 새로운 행의 인덱스 (실제로는 마지막 행 + 1)
      "나만의",
      mission, // 원본 'mission' (학생명_인증항목, 줄 바꿈 포함)을 그대로 저장
      30, 20, 10
    ]);

    const cache = CacheService.getScriptCache();
    const cacheKey = "missions_" + name;
    cache.remove(cacheKey);

    Logger.log(`'${mission}' 미션이 '학교장 인증 항목'에 추가되었습니다.`);
  } else {
    // 미션이 이미 학교장 인증 항목에 존재하는 경우 (새로운 추가가 아님)
    Logger.log(`'${mission}' 미션은 이미 '학교장 인증 항목'에 존재합니다. (정규화된 이름: ${normalizedMissionToAdd})`);
  }
}

/**
 * 특정 학생의 '학교장 인증 항목' 시트에 등록된 '나만의' 미션의 개수를 반환합니다.
 * '나만의' 미션은 '학생이름_미션내용' 형식으로 저장되어야 합니다.
 * @param {string} studentName - 개수를 셀 학생의 이름 (submitMission에서 넘어오는 'name' 값)
 * @returns {number} 해당 학생의 '나만의' 미션 개수
 */
function getCurrentCustomMissionsCount(studentName) { // 🔑 studentName 인자 추가
  const missionSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학교장 인증 항목");
  if (!missionSheet) {
    Logger.log("경고: '학교장 인증 항목' 시트를 찾을 수 없습니다.");
    return 0;
  }
  const data = missionSheet.getDataRange().getValues();

  // 🔑 헤더 행을 제외하고, 카테고리가 '나만의'이면서
  // 🔑 미션 내용 (row[2])이 해당 studentName으로 시작하는 것만 필터링하여 개수를 반환
  const count = data.filter((row, index) => {
    if (index === 0) return false; // 헤더 행 제외
    const category = row[1]; // 카테고리 (인덱스 1)
    const missionContent = row[2]; // 미션 내용 (인덱스 2)

    return category === "나만의" && String(missionContent).startsWith(`${studentName}_`);
  }).length;

  Logger.log(`학생 ${studentName}의 '나만의' 미션 개수: ${count}`); // 로그도 학생별로 표시
  return count;
}


// --- approvePointRequest 함수: 포인트 사용 요청 승인/거절 처리 (최종 완성 버전) ---
function approvePointRequest(params) {
    // 1. 함수가 어떤 데이터로 호출되었는지 기록
    Logger.log("--- approvePointRequest 함수 시작 ---");
    Logger.log("전달받은 params: " + JSON.stringify(params));

    // params 객체 자체가 비어있는지 확인
    if (!params || !params.name || !params.itemName) {
        Logger.log("CRITICAL ERROR: params 객체 또는 필수 속성(name, itemName)이 누락되었습니다.");
        return jsonResponse({ status: "error", message: "서버 오류: 요청 데이터가 유효하지 않습니다." });
    }

    const { name, itemName, approve } = params;

    try {
        Logger.log(`처리 시작: 학생='${name}', 항목='${itemName}', 승인여부=${approve}`);
        const requestSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("포인트사용요청");

        if (!requestSheet) {
            Logger.log("CRITICAL ERROR: '포인트사용요청' 시트를 찾을 수 없습니다.");
            return jsonResponse({ status: "error", message: "'포인트사용요청' 시트를 찾을 수 없습니다." });
        }

        const requests = requestSheet.getDataRange().getValues();
        Logger.log(`'포인트사용요청' 시트에서 총 ${requests.length - 1}개의 요청 데이터를 읽었습니다.`);

        let requestFound = false;

        for (let i = 1; i < requests.length; i++) {
            const row = requests[i];
            const rowStudentName = String(row[0]).trim();
            const rowItemName = String(row[1]).trim();
            const rowStatus = String(row[4]).trim();

            if (rowStudentName === name && rowItemName === itemName && rowStatus === "대기") {
                requestFound = true;
                const requestRowIndex = i + 1;
                Logger.log(`${requestRowIndex}번째 행에서 일치하는 '대기' 상태 요청을 찾았습니다.`);

                if (approve === true || approve === "true") {
                    Logger.log("승인 로직을 시작합니다...");

                    // ▼▼▼▼▼ 원래의 포인트 확인 로직 복원 ▼▼▼▼▼
                    const requestedPoint = Number(row[2]);
                    const totalEarnedPoints = getStudentTotalPoints(name);
                    Logger.log(`'${name}' 학생의 현재 총 획득 포인트: ${totalEarnedPoints}점`);

                    const alreadyApprovedUsedPoints = requests.slice(1).filter(r =>
                        String(r[0]).trim() === name &&
                        String(r[4]).trim() === "승인"
                    ).reduce((acc, cur) => acc + (Number(cur[2]) || 0), 0);
                    Logger.log(`'${name}' 학생의 기존에 승인된 사용 포인트 합계: ${alreadyApprovedUsedPoints}점`);
                    
                    const potentialTotalUsedPoints = alreadyApprovedUsedPoints + requestedPoint;
                    Logger.log(`이번 요청 포함 시 총 사용 예정 포인트: ${potentialTotalUsedPoints}점`);

                    if (totalEarnedPoints < potentialTotalUsedPoints) {
                        Logger.log(`포인트 부족으로 자동 거절됩니다. (획득: ${totalEarnedPoints} < 사용예정: ${potentialTotalUsedPoints})`);
                        requestSheet.getRange(requestRowIndex, 5).setValue("포인트부족거절");
                        return jsonResponse({
                            status: "fail",
                            message: `학생의 포인트가 부족하여 요청을 승인할 수 없습니다.`
                        });
                    }

                    requestSheet.getRange(requestRowIndex, 5).setValue("승인");
                    Logger.log(`'${name}' 학생의 '${itemName}' 요청을 '승인'으로 변경했습니다.`);
                    // ▲▲▲▲▲ 원래의 포인트 확인 로직 복원 ▲▲▲▲▲

                } else {
                    Logger.log("거절 로직을 시작합니다...");
                    requestSheet.getRange(requestRowIndex, 5).setValue("거절");
                    Logger.log(`'${name}' 학생의 '${itemName}' 요청을 '거절'으로 변경했습니다.`);
                }
                
                Logger.log("--- approvePointRequest 함수 정상 종료 (요청 처리 완료) ---");
                return jsonResponse({ status: "success", message: "처리가 완료되었습니다." });
            }
        }

        if (!requestFound) {
            Logger.log(`경고: '${name}' 학생의 '${itemName}' 항목에 대한 '대기' 상태의 요청을 찾지 못했습니다.`);
            return jsonResponse({ status: "error", message: "해당하는 대기 중인 요청을 찾을 수 없습니다." });
        }

    } catch (error) {
        Logger.log(`CRITICAL ERROR in approvePointRequest: ${error.stack}`);
        return jsonResponse({ status: "error", message: "서버 처리 중 오류 발생: " + error.message });
    }
}

// --- getPendingRequests 함수: 교사용 - 대기 중인 포인트 사용 요청 조회 ---
function getPendingRequests(teacherName) {
    Logger.log("getPendingRequests called with teacherName: '" + teacherName + "'");

    const requestSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("포인트사용요청");
    const studentSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학생 목록"); // 학년(0), 반(1), 이름(2), 파일ID(3), 담임(4)

    let requestData = [];
    if (requestSheet && requestSheet.getLastRow() > 1) {
        requestData = requestSheet.getDataRange().getValues().slice(1);
    }

    const studentInfoMap = {}; // { '학생이름': { grade: '학년', homeroomTeacher: '담임' } }
    if (studentSheet && studentSheet.getLastRow() > 1) {
        studentSheet.getDataRange().getValues().slice(1).forEach(row => {
            const grade = String(row[0]).trim(); // 학년 (인덱스 0)
            const name = String(row[2]).trim(); // 🔑 수정: 이름 (인덱스 2)
            const homeroomTeacher = String(row[4]).trim(); // 🔑 수정: 담임 (인덱스 4)
            if (name) {
                studentInfoMap[name] = { grade: grade, homeroomTeacher: homeroomTeacher };
            }
        });
    }
    Logger.log("Student Info Map generated: " + JSON.stringify(studentInfoMap));

    const filteredRequests = requestData.filter(row => {
        const studentNameInRequest = String(row[0]).trim(); // 요청한 학생의 이름 (A열, 인덱스 0)
        const requestStatus = String(row[4]).trim(); // 요청 상태 (E열, 인덱스 4)

        const studentDetailedInfo = studentInfoMap[studentNameInRequest];

        return studentDetailedInfo &&
               studentDetailedInfo.homeroomTeacher === teacherName &&
               requestStatus === "대기";
    });

    const result = filteredRequests.map(row => ({
        name: String(row[0]).trim(),
        itemName: String(row[1]).trim(),
        point: Number(row[2]) || 0,
        requestedAt: String(row[3]).trim(),
        status: String(row[4]).trim(),
        grade: studentInfoMap[String(row[0]).trim()] ? studentInfoMap[String(row[0]).trim()].grade : "학년정보없음"
    }));
    Logger.log("Final Filtered Results: " + JSON.stringify(result));

    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
}

// --- uploadImage 함수: 이미지/영상 파일을 Google Drive에 업로드 ---
function uploadImage({ name, base64 }) {
  try {
    // ⭐⭐ 여기에 디버깅 로그 추가 ⭐⭐
    Logger.log("BACKEND DEBUG: Received base64 string length: " + base64.length);
    Logger.log("BACKEND DEBUG: Received base64 string starts with: " + base64.substring(0, 100));
    const rootFolderId = "1AQmeXPXFFMwMl3dypZ_ZsKq4Y3Cienar"; // ⭐ 이 ID는 실제 Google Drive 폴더 ID로 유지해주세요 ⭐
    const rootFolder = DriveApp.getFolderById(rootFolderId);

    let studentFolder;
    const folders = rootFolder.getFoldersByName(name);
    
    if (folders.hasNext()) {
      studentFolder = folders.next();
    } else {
      studentFolder = rootFolder.createFolder(name);
    }

    // Base64 데이터에서 MIME 타입과 실제 데이터 분리
    const parts = base64.split(';base64,');
    if (parts.length !== 2) {
      throw new Error('올바른 base64 형식이 아닙니다. 파일 데이터가 손상되었을 수 있습니다.');
    }
    const mimeTypeInfo = parts[0]; // 예: "data:image/png", "data:video/mp4"
    const base64ActualData = parts[1];

    // MIME 타입에서 확장자 추출 및 안전한 기본값 설정
    const mimeType = mimeTypeInfo.split(':')[1]; // 예: "image/png", "video/mp4"
    let extension = '';
    
    // ⭐ 파일 MIME 타입에 따라 확장자 결정 로직 강화 ⭐
    if (mimeType.startsWith('image/')) {
        extension = mimeType.split('/')[1]; // image/jpeg -> jpeg
        if (extension === 'jpeg') extension = 'jpg'; // jpeg는 jpg로 통일
    } else if (mimeType.startsWith('video/')) {
        extension = mimeType.split('/')[1]; // video/mp4 -> mp4
    } else if (mimeType.startsWith('audio/')) {
        extension = mimeType.split('/')[1]; // audio/mpeg -> mpeg
    } else {
        // 알 수 없는 MIME 타입일 경우를 대비한 기본 확장자 (가장 흔한 이미지 포맷)
        Logger.log(`경고: 알 수 없는 MIME 타입 '${mimeType}'. 기본 확장자 'bin'으로 저장됩니다.`);
        extension = 'bin'; // 또는 'dat' 등 범용 확장자
    }
    
    // 파일 이름 생성 (이름_타임스탬프.확장자)
    const timestamp = new Date().getTime();
    const fileName = `${name}_${timestamp}.${extension}`;
    
    // Blob 생성 시 정확한 MIME 타입과 파일 이름을 전달
    const blob = Utilities.newBlob(Utilities.base64Decode(base64ActualData), mimeType, fileName);

    // 학생 폴더 안에 파일 생성
    const file = studentFolder.createFile(blob);

    Logger.log(`'${name}' 학생의 파일 '${fileName}' (${mimeType})이(가) Google Drive에 업로드되었습니다.`);
    return ContentService.createTextOutput(JSON.stringify({ url: file.getUrl() }))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    Logger.log(`파일 업로드 오류: ${error.message}`);
    return ContentService.createTextOutput(JSON.stringify({ error: `파일 업로드 중 오류 발생: ${error.message}` }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}

// --- logStudentLogin 함수: 학생 로그인 기록 (하루 1회) ---
function logStudentLogin(studentName) {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("로그인기록");
    const studentListSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학생 목록"); // 학년(0), 반(1), 이름(2), 파일ID(3), 담임(4)

    if (!sheet || !studentListSheet) {
        return false;
    }

    const today = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd");
    let lastRow = sheet.getLastRow();
    let data = [];

    if (lastRow > 1) {
        // 로그인기록 시트: 학년(0), 이름(1), 날짜(2), 포인트(3)
        // 이름(인덱스 1), 날짜(인덱스 2)만 가져옴
        data = sheet.getRange(2, 2, lastRow - 1, 2).getValues();
    }

    const alreadyLoggedInToday = data.some(row =>
        String(row[0]).trim() === String(studentName).trim() && Utilities.formatDate(row[1], "Asia/Seoul", "yyyy-MM-dd") === today
    );

    if (!alreadyLoggedInToday) {
        const loginTime = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");

        let studentGrade = "";
        const studentData = studentListSheet.getDataRange().getValues().slice(1);
        // 🔑 수정: 학생 목록 시트에서 '이름'(인덱스 2)으로 학생 정보 찾기
        const studentInfo = studentData.find(row => String(row[2]).trim() === String(studentName).trim());
        if (studentInfo) {
            studentGrade = studentInfo[0]; // '학년'은 학생 정보의 첫 번째 요소 (인덱스 0)
        }

        const pointSettings = getPointSettings();
        const POINT_LOGIN = pointSettings.로그인 || 1;

        // 로그인기록 시트에 학년, 이름, 날짜, 포인트 순서로 정확히 추가
        sheet.appendRow([studentGrade, studentName, loginTime, POINT_LOGIN]);
        return true;
    } else {
        return false;
    }
}

// --- getPointSettings 함수: '포인트 설정' 시트에서 포인트 기준 불러오기 ---
function getPointSettings() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("포인트 설정");
  if (!sheet) {
    return { 금: 5, 은: 3, 동: 1, 로그인: 1, 제출: 1 }; // '제출'을 한글로 유지
  }

  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return { 금: 5, 은: 3, 동: 1, 로그인: 1, 제출: 1 }; // '제출'을 한글로 유지
  }

  const settings = {};
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const grade = row[0]; // 등급 (예: '금', '은', '동', '로그인', '제출')
    const point = parseFloat(row[1]); // 포인트 값

    if (grade && !isNaN(point)) {
      settings[grade] = point; // ⭐ 수정: toLowerCase().replace(' ', '') 제거. 한글 키 그대로 사용 ⭐
    }
  }
  return settings;
}

// --- getCertificationCriteria 함수: '학교장 인증 항목' 시트에서 기준 횟수 불러오기 (수정됨) ---
function getCertificationCriteria() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const criteriaSheet = ss.getSheetByName('학교장 인증 항목');

  if (!criteriaSheet || criteriaSheet.getLastRow() <= 1) {
    return {};
  }

  const cache = CacheService.getScriptCache();
  let criteriaJson = cache.get('certificationCriteria');

  if (criteriaJson) {
    return JSON.parse(criteriaJson);
  }

  const data = criteriaSheet.getRange(2, 2, criteriaSheet.getLastRow() - 1, 5).getValues();
  const criteria = {};
  data.forEach(row => {
    const category = String(row[0]).trim(); // B열: 사업명
    const missionContent = String(row[1]).trim(); // C열: 인증 항목 내용

    // ⭐⭐⭐ 여기 조건을 수정: '나만의' 미션도 포함합니다. ⭐⭐⭐
    if (missionContent) { 
      // ⭐ '나만의' 미션의 이름은 '학생명_내용' 형태이며, 줄바꿈이 있을 수 있으므로 정규화하여 키로 사용합니다.
      const normalizedMissionKey = missionContent.replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();

      criteria[normalizedMissionKey] = { // 정규화된 미션 이름을 키로 사용
        gold: Number(row[2]) || 0, // D열: 금
        silver: Number(row[3]) || 0, // E열: 은
        bronze: Number(row[4]) || 0  // F열: 동
      };
    }
  });

  cache.put('certificationCriteria', JSON.stringify(criteria), 3600);
  return criteria;
}

// --- getStudentPointRequests 함수: 특정 학생의 포인트 사용 요청 내역 조회 ---
function getStudentPointRequests(studentName) {
  Logger.log("getStudentPointRequests called for student: '" + studentName + "'"); 

  if (!studentName) {
    Logger.log("Error: Student name is missing for getStudentPointRequests.");
    return ContentService.createTextOutput(JSON.stringify({ error: "학생 이름이 필요합니다." })).setMimeType(ContentService.MimeType.JSON);
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const requestSheet = ss.getSheetByName('포인트사용요청');
  
  if (!requestSheet || requestSheet.getLastRow() <= 1) {
    Logger.log("No point usage requests found or sheet is empty/missing for this student.");
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }

  // '포인트사용요청' 시트 열: 이름(0), 항목명(1), 필요포인트(2), 요청시각(3), 상태(4)
  const requestData = requestSheet.getRange(2, 1, requestSheet.getLastRow() - 1, 5).getValues();
  Logger.log("Total requestData rows from sheet: " + requestData.length);

  // 학생 이름으로만 필터링합니다. (대기, 승인, 거절 모두 포함)
  const filtered = requestData.filter(row => {
    const nameInSheet = String(row[0]).trim(); // 요청한 학생의 이름 (A열, 인덱스 0)
    return nameInSheet === String(studentName).trim(); 
  }).map(row => ({
    itemName: String(row[1]).trim(),
    point: Number(row[2]) || 0,
    requestedAt: Utilities.formatDate(new Date(row[3]), "Asia/Seoul", "yyyy. M. d. HH:mm"),
    status: String(row[4]).trim()
  }));
  
  // 모든 내역을 보여줄 것이므로, 최신순으로 정렬해주는 것이 좋습니다.
  filtered.sort((a, b) => new Date(b.requestedAt) - new Date(a.requestedAt));

  Logger.log("Final filtered requests count for student " + studentName + ": " + filtered.length); 
  Logger.log("Final filtered requests data for student " + studentName + ": " + JSON.stringify(filtered));

  return ContentService.createTextOutput(JSON.stringify(filtered)).setMimeType(ContentService.MimeType.JSON);
}

// --- getSchoolLogoUrl 함수: '학교 이미지' 시트에서 학교 로고 URL 가져오기 ---
function getSchoolLogoUrl() {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학교 이미지"); // ⭐ 시트 이름 확인 ⭐

    if (!sheet || sheet.getLastRow() <= 1) {
        // 시트가 없거나 데이터가 없으면 빈 URL 반환 (경고 로그 남김)
        Logger.log("경고: '학교 이미지' 시트를 찾을 수 없거나 데이터가 없습니다.");
        return ContentService.createTextOutput(JSON.stringify({ url: "" })).setMimeType(ContentService.MimeType.JSON);
    }

    const data = sheet.getDataRange().getValues().slice(1); // 헤더 제외

    // ⭐ '학교 이미지' 시트: 항목명(A열, 인덱스 0), 파일ID(B열, 인덱스 1)로 가정 ⭐
    const logoRow = data.find(row => String(row[0]).trim() === "로고"); // 항목명이 "로고"인 행 찾기

    if (logoRow && logoRow[1]) { // 항목을 찾았고 파일ID(B열, 인덱스 1)가 있다면
        const fileId = String(logoRow[1]).trim();
        const thumbnailUrl = `https://drive.google.com/thumbnail?id=${fileId}`;
        Logger.log("학교 로고 URL: " + thumbnailUrl);
        return ContentService.createTextOutput(JSON.stringify({ url: thumbnailUrl })).setMimeType(ContentService.MimeType.JSON);
    } else {
        // "로고" 항목을 찾지 못하거나 파일ID가 비어있을 경우
        Logger.log("경고: '학교 이미지' 시트에서 '로고' 항목 또는 파일 ID를 찾을 수 없습니다.");
        return ContentService.createTextOutput(JSON.stringify({ url: "" })).setMimeType(ContentService.MimeType.JSON);
    }
}

// --- 특정 학생의 최신 피드백 가져오는 함수 ---
function getLatestFeedbackForStudent(studentName, studentGrade) {
  try {
    const feedbackSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("학생피드백");
    if (!feedbackSheet) {
      return ContentService.createTextOutput(JSON.stringify({})).setMimeType(ContentService.MimeType.JSON);
    }

    const allFeedback = feedbackSheet.getDataRange().getValues().slice(1);

    const studentFeedbacks = allFeedback.filter(row => {
      const sheetGrade = String(row[0]).trim();
      const sheetName = String(row[1]).trim();
      const targetGrade = String(studentGrade).trim();
      const targetName = String(studentName).trim();

      Logger.log(`학년 비교: 시트(${sheetGrade}) vs 요청(${targetGrade})`);
      Logger.log(`이름 비교: 시트(${sheetName}) vs 요청(${targetName})`);

      return sheetGrade === targetGrade && sheetName === targetName;
    });

    if (studentFeedbacks.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({})).setMimeType(ContentService.MimeType.JSON);
    }

    studentFeedbacks.sort((a, b) => new Date(b[3]) - new Date(a[3]));
    const latestFeedback = studentFeedbacks[0];

    const result = {
      feedback: latestFeedback[2],
      timestamp: new Date(latestFeedback[3]).toLocaleString('ko-KR')
    };

    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (e) {
    return ContentService.createTextOutput(JSON.stringify({ error: e.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

// --- getCertificationExplanations 함수: 인증 항목 설명 자료 조회 ---
function getCertificationExplanations() {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("인증 항목 설명"); // ⭐ 시트 이름 확인 ⭐
    if (!sheet || sheet.getLastRow() <= 1) {
        Logger.log("경고: '인증 항목 설명' 시트를 찾을 수 없거나 데이터가 없습니다.");
        return jsonResponse([]);
    }

    // A열(사업명)부터 D열(사진)까지 4개 열을 가져옵니다.
    const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues(); 
    const explanations = data.map(row => {
        const category = String(row[0] || '').trim();
        const missionItem = String(row[1] || '').trim();
        const description = String(row[2] || '').trim();
        const rawImageUrl = String(row[3] || '').trim(); // Google Drive 파일 ID 또는 URL

        let displayImageUrl = '';
        if (rawImageUrl) {
             // rawImageUrl이 http로 시작하면 직접 URL로 사용, 아니면 Drive 파일 ID로 간주하여 썸네일 URL 생성
             if (rawImageUrl.startsWith('http')) {
                 displayImageUrl = rawImageUrl;
             } else {
                 displayImageUrl = `https://drive.google.com/thumbnail?id=${rawImageUrl}`;
             }
        }

        return {
            category: category,
            missionItem: missionItem,
            description: description,
            imageUrl: displayImageUrl // HTML에서 바로 사용할 수 있는 URL
        };
    });
    Logger.log(`인증 항목 설명 ${explanations.length}개 로드 완료.`);
    return jsonResponse(explanations);
}

// --- updateSubmissionGrades() 함수: 모든 학생의 제출 내역에 등급 자동 부여 (수정됨) ---
function updateSubmissionGrades() {
    const submissionsSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("제출 내역");
    const certificationCriteria = getCertificationCriteria(); // "학교장 인증 항목"에서 기준을 가져옴
    
    if (!submissionsSheet) {
        Logger.log("오류: '제출 내역' 시트를 찾을 수 없어 등급 업데이트 불가.");
        return;
    }

    const submissionData = submissionsSheet.getDataRange().getValues(); 
    if (submissionData.length <= 1) { // 헤더만 있거나 데이터가 없으면 종료
        Logger.log("정보: '제출 내역' 시트에 업데이트할 제출 데이터가 없습니다.");
        return;
    }
    const headers = submissionData[0]; 
    let dataRows = submissionData.slice(1); // 실제 데이터 행들

    // '제출 내역' 시트 컬럼 인덱스 확인 (헤더를 기반으로 안전하게 찾기)
    const NAME_COL = headers.indexOf("이름"); 
    const CATEGORY_COL = headers.indexOf("사업명"); // 사용되지는 않지만, 혹시 모를 상황 대비
    const MISSION_COL = headers.indexOf("인증항목");
    const SUBMISSION_TIME_COL = headers.indexOf("제출시각"); // 제출 시각 컬럼 인덱스
    const LEVEL_COL = headers.indexOf("등급"); // 등급 컬럼 인덱스

    if (NAME_COL === -1 || CATEGORY_COL === -1 || MISSION_COL === -1 || SUBMISSION_TIME_COL === -1 || LEVEL_COL === -1) {
        Logger.log("오류: '제출 내역' 시트의 필수 컬럼(이름, 사업명, 인증항목, 제출시각, 등급)을 찾을 수 없어 등급 업데이트 불가.");
        return;
    }

    // ⭐ 제출 데이터를 제출시각 기준으로 오름차순 정렬 (시간 순서대로 처리하기 위함) ⭐
    // 원본 인덱스를 함께 저장하여 나중에 등급을 올바른 위치에 적용할 수 있도록 함
    dataRows = dataRows.map((row, index) => ({ row: row, originalIndex: index }));
    dataRows.sort((a, b) => {
        const timeA = new Date(a.row[SUBMISSION_TIME_COL]);
        const timeB = new Date(b.row[SUBMISSION_TIME_COL]);
        return timeA.getTime() - timeB.getTime();
    });

    const studentMissionCurrentCounts = {}; // 각 학생-미션별 현재까지의 누적 제출 횟수
    const studentMissionAwardedGrades = {}; // 각 학생-미션별로 어떤 등급이 이미 부여되었는지 추적

    // 최종적으로 시트에 적용할 새로운 등급 값들을 저장할 배열 (원본 데이터 행 수만큼 초기화)
    // 모든 등급을 빈 문자열로 초기화하여, 등급이 부여되지 않는 행은 비워두도록 함
    const newGradesToApply = Array(dataRows.length).fill(null).map(() => [""]); 

    for (let i = 0; i < dataRows.length; i++) {
        const { row, originalIndex } = dataRows[i]; // 정렬된 행과 그 행의 원본 인덱스

        const studentName = String(row[NAME_COL]).trim();
        const missionName = String(row[MISSION_COL]).trim(); // '학생이름_실제항목명' 형태일 수 있음

        // 학생 및 미션별 추적 객체 초기화
        if (!studentMissionCurrentCounts[studentName]) {
            studentMissionCurrentCounts[studentName] = {};
            studentMissionAwardedGrades[studentName] = {};
        }
        if (!studentMissionCurrentCounts[studentName][missionName]) {
            studentMissionCurrentCounts[studentName][missionName] = 0;
            studentMissionAwardedGrades[studentName][missionName] = { bronze: false, silver: false, gold: false };
        }

        // 해당 미션의 누적 제출 횟수 증가
        studentMissionCurrentCounts[studentName][missionName]++;
        const currentCumulativeCount = studentMissionCurrentCounts[studentName][missionName];

        // 해당 미션의 등급 기준을 가져옴 (정규화된 미션 이름으로 조회)
        const normalizedMissionKey = missionName.replace(/\n/g, ' ').replace(/\s+/g, ' ').trim();
        const criteria = certificationCriteria[normalizedMissionKey]; 
        
        let gradeForThisSubmission = ""; // 이 특정 제출에 부여할 등급 (기본값은 빈 문자열)

        if (criteria) {
            const awardedGrades = studentMissionAwardedGrades[studentName][missionName];

            // ⭐ 금 등급 확인 (가장 높은 우선순위) ⭐
            // 현재 누적 횟수가 금장 기준과 일치하고, 아직 금장이 부여되지 않았다면
            if (currentCumulativeCount === criteria.gold && !awardedGrades.gold) {
                gradeForThisSubmission = "금";
                awardedGrades.gold = true; // 금장 부여됨 표시
                // 금장을 받으면 은장, 동장도 자동으로 '부여됨'으로 간주하여 중복 방지
                awardedGrades.silver = true; 
                awardedGrades.bronze = true; 
            } 
            // ⭐ 은 등급 확인 ⭐
            // 현재 누적 횟수가 은장 기준과 일치하고, 아직 은장이 부여되지 않았다면
            else if (currentCumulativeCount === criteria.silver && !awardedGrades.silver) {
                gradeForThisSubmission = "은";
                awardedGrades.silver = true; // 은장 부여됨 표시
                // 은장을 받으면 동장도 자동으로 '부여됨'으로 간주하여 중복 방지
                awardedGrades.bronze = true; 
            } 
            // ⭐ 동 등급 확인 ⭐
            // 현재 누적 횟수가 동장 기준과 일치하고, 아직 동장이 부여되지 않았다면
            else if (currentCumulativeCount === criteria.bronze && !awardedGrades.bronze) {
                gradeForThisSubmission = "동";
                awardedGrades.bronze = true; // 동장 부여됨 표시
            }
        }
        
        // 이 특정 제출의 원본 위치에 계산된 등급을 저장 (등급이 없으면 빈 문자열)
        newGradesToApply[originalIndex][0] = gradeForThisSubmission;
    }

    // 계산된 모든 등급을 '제출 내역' 시트에 한 번에 업데이트
    if (newGradesToApply.length > 0) {
        submissionsSheet.getRange(2, LEVEL_COL + 1, newGradesToApply.length, 1).setValues(newGradesToApply);
        Logger.log(`'제출 내역' 시트의 등급 ${newGradesToApply.length}개 행이 업데이트되었습니다. (새로운 로직 적용)`);
    } else {
        Logger.log("정보: '제출 내역' 시트에 등급 업데이트할 내용이 없습니다.");
    }
}

// 🎯 이모지 시트 설정을 위한 변수 (본인 환경에 맞게 수정!)
const EMOJI_SHEET_NAME = "이모지";

/**
 * '이모지' 시트에서 목록을 읽어와 JSON 형태로 반환하는 함수 (기존 방식 적용)
 */
function getEmojis() {
  try {
    // 🎯 ID 대신 getActiveSpreadsheet()를 사용하는 방식으로 변경
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(EMOJI_SHEET_NAME);
    
    if (!sheet) {
      throw new Error(`'${EMOJI_SHEET_NAME}' 시트를 찾을 수 없습니다.`);
    }

    // 데이터가 A2부터 시작한다고 가정. A열: 이모지, B열: 포인트
    const range = sheet.getRange("A2:B" + sheet.getLastRow());
    const values = range.getValues();
    
    const emojis = values
      .filter(row => row[0] && row[0].toString().trim() !== "") // 이모지 칸이 비어있지 않은 행만 필터링
      .map(row => ({
        emoji: row[0], // A열
        point: row[1]  // B열
      }));

    return ContentService.createTextOutput(JSON.stringify(emojis))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    Logger.log(error.message); // 로그에 에러 기록
    const errorResponse = { status: "error", message: error.message };
    return Content-Service.createTextOutput(JSON.stringify(errorResponse))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * [접수 담당] 프론트엔드로부터 이모지 추가 요청을 받아 처리하는 함수
 */
function addEmoji(params) {
  const { name, emojiIndex, point } = params;
  if (name === undefined || emojiIndex === undefined || point === undefined) {
    return jsonResponse({ status: "error", message: "필수 데이터(이름, 이모지, 포인트)가 누락되었습니다." });
  }
  // 실제 로직을 처리하는 실무 담당 함수 호출
  return addEmojiToProfile(name, emojiIndex); 
}


/**
 * [실무 담당] 학생 프로필에 이모지를 누적하고 포인트를 차감하는 함수
 */
function addEmojiToProfile(studentName, emojiIndex) {
    // getEmojis() 함수는 이미 Code.gs에 있어야 합니다.
    const emojiListResponse = getEmojis(); 
    const emojis = JSON.parse(emojiListResponse.getContent());

    if (emojiIndex >= emojis.length) {
        return jsonResponse({ status: "error", message: "유효하지 않은 이모지 인덱스입니다." });
    }
    const selectedEmoji = emojis[emojiIndex];
    const requiredPoint = selectedEmoji.point;

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const studentsSheet = ss.getSheetByName("학생 목록");
    if (!studentsSheet) {
        return jsonResponse({ status: "error", message: "학생 목록 시트를 찾을 수 없습니다." });
    }

    const studentData = studentsSheet.getDataRange().getValues();
    const header = studentData[0];
    const nameCol = header.indexOf('이름');
    const emojiCol = header.indexOf('이모지');

    if (nameCol === -1 || emojiCol === -1) {
        return jsonResponse({ status: "error", message: "'학생 목록' 시트에 '이름' 또는 '이모지' 열이 없습니다." });
    }

    let studentRowIndex = -1;
    for (let i = 1; i < studentData.length; i++) {
        if (studentData[i][nameCol] === studentName) {
            studentRowIndex = i + 1; // 1-based index
            break;
        }
    }

    if (studentRowIndex === -1) {
        return jsonResponse({ status: "error", message: "학생을 찾을 수 없습니다." });
    }

    // 학생의 현재 사용 가능 포인트를 정확히 다시 계산
    const statsContent = getStudentStats(studentName).getContent();
    const currentAvailablePoints = JSON.parse(statsContent).availablePoints || 0;

    if (currentAvailablePoints < requiredPoint) {
        return jsonResponse({ status: "error", message: `포인트가 부족합니다. (현재 ${currentAvailablePoints}점)` });
    }

    // 포인트 사용 내역 기록
    const pointUsageSheet = ss.getSheetByName('포인트사용요청');
    if (pointUsageSheet) {
        const timestamp = new Date();
        pointUsageSheet.appendRow([
            studentName,
            `이모지 구매: ${selectedEmoji.emoji}`,
            requiredPoint,
            timestamp,
            '승인' // 이모지 구매는 즉시 승인 처리
        ]);
    }
    
    // 이모지 업데이트 (최대 2개)
    const currentEmojiString = studentsSheet.getRange(studentRowIndex, emojiCol + 1).getValue().toString() || "";
    let emojiArray = currentEmojiString ? currentEmojiString.split(',').map(e => e.trim()) : [];

    emojiArray.push(selectedEmoji.emoji);

    if (emojiArray.length > 2) {
        emojiArray.shift(); // 가장 오래된 이모지 삭제
    }

    const newEmojiString = emojiArray.join(', ');
    studentsSheet.getRange(studentRowIndex, emojiCol + 1).setValue(newEmojiString);
    
    return jsonResponse({ status: "success", message: "이모지가 추가되었습니다!" });
}

/**
 * 학생의 포인트 사용 요청을 '포인트사용요청' 시트에 기록하는 함수
 */
function requestPoint(params) {
  try {
    const { name, itemName, point } = params;

    // 필수 정보가 있는지 확인
    if (!name || !itemName || point === undefined) {
      return jsonResponse({ status: "error", message: "요청에 필요한 정보(이름, 항목, 포인트)가 부족합니다." });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const requestSheet = ss.getSheetByName("포인트사용요청");
    if (!requestSheet) {
      throw new Error("'포인트사용요청' 시트를 찾을 수 없습니다.");
    }
    
    // 동일한 학생이 동일한 항목을 '대기' 상태로 이미 요청했는지 확인 (중복 요청 방지)
    const data = requestSheet.getDataRange().getValues();
    const hasPendingRequest = data.some(row => 
      row[0] === name &&       // 이름 (A열)
      row[1] === itemName &&   // 항목명 (B열)
      row[4] === "대기"        // 상태 (E열)
    );

    if (hasPendingRequest) {
      return jsonResponse({ status: "fail", message: "이미 동일한 항목을 요청했으며, 선생님의 승인을 기다리는 중입니다." });
    }

    const timestamp = new Date();
    
    // '포인트사용요청' 시트에 새 행으로 추가
    // [이름, 항목명, 필요포인트, 요청시각, 상태]
    requestSheet.appendRow([name, itemName, point, timestamp, "대기"]);

    Logger.log(`'${name}' 학생의 '${itemName}' (${point}점) 포인트 사용 요청이 기록되었습니다.`);
    return jsonResponse({ status: "success", message: "포인트 사용 요청이 완료되었습니다." });

  } catch (error) {
    Logger.log("requestPoint 함수 오류: " + error.stack);
    return jsonResponse({ status: "error", message: "서버 내부 오류: " + error.message });
  }
}
